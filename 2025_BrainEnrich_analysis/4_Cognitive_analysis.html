# Load hcp brain_data
gene_data <- get_geneExp(atlas = "desikan", rdonor = "r0.4", hem = "L")

var2test <- colnames(behav_df)

iter_total <- 5000 # Total number of iterations
cor_method <- "pearson" # Correlation method
minGSSize <- 1 # Minimum gene set size
maxGSSize <- 5000 # Maximum gene set size

# Define an array of null models, gene set types, and aggregation methods to loop through
null_models <- c("spin_brain")
gs_types <- c("SynGO")
aggre_methods <- c("ks_pos_neg_sum")

res_list<-list()
# Loop through each combination of null model, gene set type, and aggregation method
for (null_model in null_models) {
  for (gs_type in gs_types) {
    annoData <- get_annoData(type = gs_type)
    for (aggre_method in aggre_methods) {
      print(paste(gs_type, null_model, cor_method, aggre_method, minGSSize, maxGSSize, iter_total, sep = "_"))
      # Construct the folder name based on the parameters
      nllScoreRds <- sprintf(
        "%s/%s_%s_%s_%s_%d_%d_%d.rds",
        null_score_dir, gs_type, null_model, cor_method, aggre_method, minGSSize, maxGSSize, iter_total
      )
      nullscore <- readRDS(nllScoreRds)
      for (var_i in var2test){
      message(sprintf("Processing: %s - %s - %s - %s (%s)", gs_type, null_model, cor_method, aggre_method, var_i))

      pred_df<-behav_df %>% dplyr::select(all_of(var_i))
      res <- brainscore.lm_test(
        pred_df = pred_df,
        cov_df = cov_df,
        brain_data = brain_data,
        gene_data = gene_data,
        annoData = annoData,
        gsScoreList.null = nullscore,
        cor_method = "pearson",
        aggre_method = aggre_method,
        n_cores = 15,
        minGSSize = minGSSize,
        maxGSSize = maxGSSize,
        null_model = null_model,
        n_perm = iter_total,
        pvalueCutoff = 1,
        gsea_obj = FALSE, # return a df instead of a gsea object
        threshold_type = "none"
      )  
      res_list[[var_i]] <- res
      
      }

    saveRDS(res_list, file.path(res_path, sprintf("res_ohter_variables_List_%s_%s_%s_%s_%d_%d_%d.rds", gs_type, null_model, cor_method, aggre_method, minGSSize, maxGSSize, iter_total)),compress='xz')
    }
  }
}

)

# Extract selected predictors
selected_predictors <- names(predictor_labels)

# Clean and prepare the `Description` column
cleaned_df <- res_df %>%
  mutate(
    # Remove text within parentheses
    Description = gsub("\\(.*?\\)", "", Description),
    # Wrap text for better visualization
    Description = sapply(Description, function(x) {
      paste(strwrap(x, width = 40, simplify = TRUE), collapse = "\n")
    })
  )

  filter(Predictor %in% selected_predictors)
# save table S6  
#write.csv(cleaned_df2save, file.path(res_path, 'selected_cognition_measures_spin_brain_sorted.csv'), row.names = FALSE)


# Count significant associations for sorting and filter out descriptions with no significant results
sig_counts <- cleaned_df %>%
  filter(Predictor %in% selected_predictors) %>%
  mutate(Significant = ifelse(np.pval < 0.05, 1, 0)) %>%
  group_by(Description) %>%
  summarise(SigCount = sum(Significant), .groups = "drop") %>%
  filter(SigCount > 0)  # Keep only rows with at least one significant result

# Determine the sorted order of descriptions
sorted_descriptions <- sig_counts %>%
  arrange(desc(SigCount)) %>%
  pull(Description)

# Prepare the final data frame for plotting
res_df_sorted <- cleaned_df %>%
  filter(
    Predictor %in% selected_predictors,
    Description %in% sorted_descriptions  # Keep only significant descriptions
  ) %>%
  mutate(
    # # Sort `Description` by the number of significant results
    Description = factor(Description, levels = sorted_descriptions),
    # Replace `Predictor` with their full labels
    Predictor = factor(Predictor, levels = selected_predictors, labels = predictor_labels)
  )
